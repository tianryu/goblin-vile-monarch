extends RefCounted
var ui
var m
var kind=""
var ident=0
func _init(owner_ui):ui=owner_ui;m=ui.model
func size():return ui.panel_content.size
func image(name,rect,parent=null):
	var p=TextureRect.new();p.texture=ui.art.texture(name);p.expand_mode=TextureRect.EXPAND_IGNORE_SIZE;p.stretch_mode=TextureRect.STRETCH_KEEP_ASPECT_CENTERED;p.mouse_filter=Control.MOUSE_FILTER_IGNORE
	p.position=rect.position;p.size=rect.size;(ui.panel_content if parent==null else parent).add_child(p);return p
func words(value,rect,font_size=20,parent=null,align=HORIZONTAL_ALIGNMENT_LEFT):
	var l=ui.label(value,ui.panel_content if parent==null else parent,font_size);l.position=rect.position;l.size=rect.size;l.horizontal_alignment=align
	l.autowrap_mode=TextServer.AUTOWRAP_WORD_SMART;l.clip_text=true;return l
func selected(default_value):
	if not ui.selections.has(kind):ui.selections[kind]=default_value
	return str(ui.selections[kind])
func choose(value):ui.selections[kind]=value;ui.open_panel(kind,ident)
func execute(callable,close=false):
	var result=callable.call();ui._save();ui._refresh()
	var ok=result.get("ok",false) if result is Dictionary else bool(result)
	if close and ok:ui.close_panel()
	else:ui.open_panel(kind,ident)
func action(value,callback,key,disabled=false,slot="primary"):return ui.mobile.footer(value,callback,key,slot,disabled)
func header(title,description="",sheet=true):ui.mobile.header(title,description,sheet)
func tier():return clampi(int(m.hall().level),1,3)
func role_art(role):return role+("_"+str(tier()) if tier()>1 else "")
func card(index,count,art,title,detail,price=-1,choice="",chosen=""):
	var available=size();var gap=14.0;var w=(available.x-gap*(count-1))/count;var h=minf(232,available.y);var x=index*(w+gap);var y=(available.y-h)/2
	var detail_h=maxf(49,ceilf(ui.theme.default_font.get_height(18))*2)
	var price_y=h-41;var detail_y=price_y-6-detail_h;var title_y=detail_y-34
	var art_rect=Rect2(12,12,w-24,maxf(20,minf(100,title_y-16)))
	var b=ui.button("",ui.panel_content,choose.bind(choice),"select_"+choice if choice!="" else "")
	b.set_meta("qa_card",true);b.set_meta("art_rect",art_rect)
	b.position=Vector2(x,y);b.size=Vector2(w,h);b.mouse_filter=Control.MOUSE_FILTER_IGNORE if choice=="" else Control.MOUSE_FILTER_STOP
	for mode in ["normal","hover","pressed","focus","disabled"]:
		var texture="ui_card_selected_v5" if choice!="" and choice==chosen else "ui_card_v5"
		var box=ui.skin.box(texture,Vector2(12,14),Vector2.ZERO);box.modulate_color=Color(0.94,0.94,0.90) if mode=="pressed" else Color.WHITE;b.add_theme_stylebox_override(mode,box)
	image(art,art_rect,b).name="CardArt"
	words(title,Rect2(10,title_y,w-20,31),23,b,HORIZONTAL_ALIGNMENT_CENTER).name="CardTitle"
	words(detail,Rect2(10,detail_y,w-20,detail_h),18,b,HORIZONTAL_ALIGNMENT_CENTER).name="CardDetail"
	if price>=0:
		image("coin",Rect2(w/2-31,price_y+4,21,21),b);words(str(price),Rect2(w/2-5,price_y,68,30),22,b).name="CardPrice"
	return b
func message(art,lines):
	var s=size();image(art,Rect2(12,12,240,s.y-24))
	var y=(s.y-lines.size()*38)/2
	for line in lines:words(line,Rect2(282,y,s.x-300,37),21);y+=38
func buy_tool(role):execute(m.buy_tool.bind(role))
func equip_mount(mount):execute(m.equip_mount.bind(mount),true)
func buy_lizard():
	if m.buy_lizard():m.equip_mount("lizard");ui.close_panel();ui._save()
	else:ui._refresh()
func sell(species):execute(m.sell_captives.bind(species))
func exchange():execute(m.sell_captives.bind("all",true))
func start_theft(target):execute(m.start_theft.bind(target),true)
func rally(side):execute(m.rally.bind(side),true)
func switch_side():ui.theft_side=-ui.theft_side;ui.open_panel("steal",ident)
func prison_upgrade():ui.close_panel();m.focus_prison_upgrade();ui._refresh()
func new_campaign():
	m.new_campaign();ui.world.camera_x=m.state.hero.x;ui.world.previous={};ui.world.effects.clear();ui.close_panel()
func guide_next():
	if ui.guide_page<3:ui.guide_page+=1;ui.open_panel("guide")
	else:ui.close_panel()
func guide_prev():ui.guide_page-=1;ui.open_panel("guide")
func pause_entry(key):
	if key=="sound":m.global_data.settings.sound=not m.global_data.settings.sound;ui._save();ui.open_panel("pause")
	elif key=="reflections":m.global_data.settings.low_fx=not m.global_data.settings.low_fx;ui._save();ui.open_panel("pause")
	else:ui.open_panel(key)
