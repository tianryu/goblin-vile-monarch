func _notification(what):
	if what==NOTIFICATION_APPLICATION_FOCUS_OUT or what==NOTIFICATION_APPLICATION_PAUSED:
		if model!=null:model.pause("background",true);store.write(model)
		_clear_input()
	elif what==NOTIFICATION_APPLICATION_FOCUS_IN or what==NOTIFICATION_APPLICATION_RESUMED:
		if model!=null:model.pause("background",false)
		_clear_input()

func _input(event):
	if event is InputEventMouse and event.device==InputEvent.DEVICE_ID_EMULATION:return
	if event is InputEventKey and not event.echo:
		if event.pressed and event.keycode==KEY_ESCAPE:
			if modal!="":handle_back()
			else:open_panel("pause")
		if event.keycode in [KEY_E,KEY_SPACE]:
			if event.pressed and modal=="":key_action_held=true;_begin_action()
			elif not event.pressed:key_action_held=false;target_lock={};action_consumed=false
	if modal!="" or rotate_cover.visible:return
	if event is InputEventScreenTouch:
		if event.pressed and not event.canceled:_press_touch(event.index,event.position)
		else:
			var canceled=event.canceled
			if OS.has_feature("web"):
				canceled=canceled or bool(JavaScriptBridge.eval("(()=>{const s=window.GoblinInputCancelled;if(!s)return false;const id="+str(event.index)+";const hit=s.has(id);s.delete(id);return hit})()",true))
			_release_touch(event.index,canceled)
			if canceled:model.cancel_payment()
	elif event is InputEventScreenDrag:
		if event.index==joystick_finger:_drag_joystick(event.position)
	elif event is InputEventMouseButton and event.button_index==MOUSE_BUTTON_LEFT:
		if event.pressed:_press_touch(-1,event.position)
		else:_release_touch(-1)
	elif event is InputEventMouseMotion and joystick_finger==-1:_drag_joystick(event.position)

func _press_touch(finger,pos):
	if joystick.get_global_rect().has_point(pos) and joystick_finger==-999:
		joystick_finger=finger;_drag_joystick(pos)
	elif run_button.get_global_rect().has_point(pos) and run_finger==-999:run_finger=finger;model.input_run=true
	elif action_button.is_visible_in_tree() and action_button.get_global_rect().has_point(pos) and action_finger==-999:
		action_finger=finger;_begin_action()

func _release_touch(finger,canceled=false):
	if finger==joystick_finger:joystick_finger=-999;model.input_axis=0;joystick.set_axis(0)
	if finger==run_finger:run_finger=-999;model.input_run=false
	if finger==action_finger:
		action_finger=-999;target_lock={};action_consumed=false
		if not pending_shop.is_empty():
			var next=pending_shop.duplicate();pending_shop={}
			var b=model.by_id(model.state.buildings,next.get("object_id",0))
			if not canceled and not b.is_empty() and abs(model.state.hero.x-model.describe_building(b).x)<=86 and model.state.hero.fly<=3:open_panel(next.shop,next.get("object_id",0))
			get_viewport().set_input_as_handled()

func _begin_action():
	target_lock=model.find_target().duplicate(true);action_interval=0.18;action_consumed=false
	if target_lock.is_empty():action_consumed=true;return
	_do_action()

func _drag_joystick(pos):
	var dx=(pos.x-joystick.get_global_rect().get_center().x)/40.0
	model.input_axis=0.0 if abs(dx)<0.18 else clampf(dx,-1,1)
	joystick.set_axis(model.input_axis)

func _do_action():
	if model.paused() or action_consumed:return
	var result=model.contribute(target_lock)
	if result.get("shop","")!="":
		if action_finger!=-999:pending_shop=result.duplicate()
		else:open_panel(result.shop,result.get("object_id",0))
	if not result.get("payment",false) or result.get("completed",false) or not result.get("ok",false):action_consumed=true
	_save();_refresh()

func _process(delta):
	if ad_inflight and Time.get_ticks_msec()>ad_deadline:_on_ad_result([],ad_request_serial)
	if joystick_finger==-999 and modal=="" and not model.paused():
		model.input_axis=float(int(Input.is_physical_key_pressed(KEY_D) or Input.is_physical_key_pressed(KEY_RIGHT))-int(Input.is_physical_key_pressed(KEY_A) or Input.is_physical_key_pressed(KEY_LEFT)))
	if run_finger==-999 and modal=="":model.input_run=Input.is_physical_key_pressed(KEY_SHIFT)
	if (action_finger!=-999 or key_action_held) and not action_consumed and modal=="":
		action_interval-=delta
		if action_interval<=0:_do_action();action_interval=0.18
	accumulator+=minf(delta,0.1)
	while accumulator>=1.0/30.0:
		world.capture_previous();model.tick(1.0/30.0);accumulator-=1.0/30.0
	world.interpolation=accumulator*30.0
	for b in [action_button,run_button]:
		var pressed=(action_finger!=-999 or key_action_held) if b==action_button else model.input_run
		b.scale=b.scale.lerp(Vector2.ONE*(0.94 if pressed else 1.0),1-exp(-delta*22))
	autosave+=delta;refresh_timer+=delta
	if autosave>=5 and model.dirty:_save();autosave=0
	if refresh_timer>0.08:
		_refresh();refresh_timer=0
		if OS.has_feature("web") and OS.is_debug_build():
			if not qa_stress_mode or Time.get_ticks_msec()>=qa_next_publish_msec:
				_publish_qa();qa_next_publish_msec=Time.get_ticks_msec()+500
	if model.state.status!="playing" and modal not in ["result","ledger","guide","credits"]:open_panel("result")

func short_number(value):
	if value>=100000000:return str(int(value/100000000))+"亿"
	if value>=10000:return str(int(value/10000))+"万"
	return str(value)

func stage_hint():
	var s=model.state
	if s.flags.left and s.flags.right:return "联军退散，森林归部落！"
	if s.flags.left or s.flags.right:return "最后一面军旗  "+("前往东营" if s.flags.left else "前往西营")
	var west=0;var east=0
	for e in s.enemies:
		if e.mode=="active" and not e.get("guard",false):
			if e.side<0:west+=1
			else:east+=1
	if west+east>0:return "联军来袭  西侧"+str(west)+"  东侧"+str(east)
	if s.total<18:return "第一步  修筑两侧路障，守住大厅"
	if s.recruited<=6 and s.day<=2:return "森林营地招募爪牙  工具摊分配职业"
	if model.building("prison").level==0 and s.kills>0:return "建造监牢  工匠会拖回战场上的活口"
	if model.building("den").level==0 and s.day>=2:return "修建盗贼窝  偷走联军物资再出征"
	return "征服西营与东营，把两面军旗带回大厅"

	elif modal=="guide":open_panel(guide_return)

func request_ad(kind):
	if ad_inflight:return
	# Live completion is only accepted from the platform adapter, never a timer or close button.
	if not OS.has_feature("web") or not JavaScriptBridge.eval("typeof window.GoblinRewardedAds === 'object' && window.GoblinRewardedAds.ready === true",true):
		model.notify("广告暂时不可用，可以继续游玩。");_refresh()
		return
	ad_inflight=true;model.pause("ad",true);_clear_input()
	panel_feedback.text="广告播放中，完成后领取奖励"
	var pending_controls=[];_append_controls(panel_layer,pending_controls)
	for c in pending_controls:
		if c is Button:c.disabled=true
	ad_requested_kind=kind
	ad_request_serial+=1;ad_deadline=Time.get_ticks_msec()+180000
	ad_callback=JavaScriptBridge.create_callback(_on_ad_result.bind(ad_request_serial))
	var window=JavaScriptBridge.get_interface("window");window.goblinAdResult=ad_callback
	JavaScriptBridge.eval("try{window.GoblinRewardedAds.show("+JSON.stringify(kind)+",window.goblinAdResult)}catch(e){window.goblinAdResult(JSON.stringify({completed:false}))}",true)

func _on_ad_result(args,serial=-1):
	if not ad_inflight or serial!=ad_request_serial:return
	ad_inflight=false;model.pause("ad",false)
	var parser=JSON.new();var data=null
	if args.size()>0 and parser.parse(str(args[0]))==OK:data=parser.data
	if data is Dictionary and data.get("completed",false) and data.get("receipt","")!="" and data.get("kind","")==ad_requested_kind:
		model.grant_reward(ad_requested_kind,data.receipt);_save()
	else:model.notify("未完成观看，没有发放奖励")
	ad_requested_kind=""
	if modal!="":open_panel(modal,modal_object)
	else:_save();_refresh()

func _qa_command(args):

	if not command is Dictionary:return

		"move":model.state.hero.x=float(command.x);world.camera_x=model.state.hero.x
		"close":close_panel()
		"new":qa_stress_mode=false;model.new_campaign(int(command.get("seed",42)),false);close_panel()
		"time":model.state.time=float(command.value)
		"reward":model.grant_reward(command.kind,command.receipt)
		"open":open_panel(command.kind,command.get("id",0))
		"resources":model.state.gold=int(command.get("gold",50));model.state.scrap=int(command.get("scrap",100))
		"guide_page":
			if modal!="guide":open_panel("guide")

		"freeze":model.pause("qa_scene",bool(command.get("value",true)))

			for u in model.state.units:

				if b.kind=="net" and b.x==1660:b.level=2;b.charges=1
			for i in range(5):model.add_unit("hunter",1415+i*15)
			for i in range(2):model.add_unit("brute",1490+i*22)
			for i in range(6):model.spawn_enemy(["human","elf","dwarf"][i%3],1,1740+i*55)

				if u.role=="hunter":model._hunter(u,0.0)
			model.pause("qa_scene",true)
		"scene":

			model.state.trader.phase="staying";model.state.trader.x=model.CORE_SITES.trader_stop;model.state.trader.until=99999
			var worker=model.state.units[0];worker.x=30;worker.face=1;worker.cargo={"kind":"captive","type":"human"}

			if command.get("interactive",false):worker.cargo={};greedy.hidden=0;model.pause("qa_scene",false)

			_qa_command([JSON.stringify({"op":"scene"})]);hud.hide();panel_layer.hide()
