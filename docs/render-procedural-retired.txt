import {GROUND} from './model.js';
const C={g:'#809b44',G:'#acbf63',d:'#405932',e:'#172c29',b:'#715d40',B:'#a18a56',t:'#374a50',r:'#944735',y:'#efc256',w:'#d1c7a0',s:'#736256',S:'#958269',p:'#593c68',P:'#9774aa',k:'#253034'};
const goblin=['  g     g  ','   gGGGg   ','  gGGeGeGg ','   gGGGg   ','    ddd    ','   bbbbb   ','  gbbbbbgg ','  g bbb  g ','    ttt    ','   tt tt   ','   bb bb   '];
const boar=['                 ','       sss       ','  sssssssss      ',' sssssSSssss  w  ','ssssSSSSsssssw w ','sssssssssssesww  ',' sssssssssssss   ','   ss     ss     ','   kk     kk     '];
const wolf=['             s   ','           sss   ','     sssssssssw  ',' sssssSSsssseww  ','ss  ssssssssss   ','    ss   ss      ','   ss     ss     '];
const lizard=['                 ','        ddd   d  ','  ddddddGGGdddGd ','dddddGGGGGGGGeGw ','   ddddddddddd   ','    dd     dd    ','   ddd    ddd    '];
const bat=['p                p',' pp    pppp    pp ',' ppp  pGGeGp  ppp','  ppppGGGGppppp  ','    pppppppp     ','       pp        '];
export class Renderer{
 constructor(canvas,game){this.canvas=canvas;this.ctx=canvas.getContext('2d',{alpha:false});this.game=game;this.camera=0;this.image=new Image();this.ready=new Promise((resolve,reject)=>{this.image.onload=resolve;this.image.onerror=()=>reject(Error('环境资源加载失败'));});this.image.src='./assets/tribe-environment.png';this.w=1774;this.h=887;this.t=0;this.fps=60;this.resize();}
 resize(){const rect=this.canvas.getBoundingClientRect();this.w=Math.round(887*rect.width/Math.max(1,rect.height));this.h=887;this.canvas.width=this.w;this.canvas.height=this.h;this.ctx.imageSmoothingEnabled=false;}
 sx(x){return x-this.camera+this.w/2;}
 pixel(rows,x,y,scale=3,face=1,palette=C){const c=this.ctx;c.save();c.translate(Math.round(x),Math.round(y));c.scale(face,1);for(let j=0;j<rows.length;j++)for(let i=0;i<rows[j].length;i++){const color=palette[rows[j][i]];if(color){c.fillStyle=color;c.fillRect(Math.round((i-rows[j].length/2)*scale),Math.round((j-rows.length)*scale),scale,scale);}}c.restore();}
 rect(x,y,w,h,color){this.ctx.fillStyle=color;this.ctx.fillRect(Math.round(x),Math.round(y),Math.round(w),Math.round(h));}
 text(text,x,y,color='#d8d7b2',size=16,align='center'){const c=this.ctx;c.fillStyle=color;c.font=`${size}px "Microsoft YaHei",sans-serif`;c.textAlign=align;c.fillText(text,Math.round(x),Math.round(y));}
 npc(n,x=this.sx(n.x),y=GROUND,reflection=false){const c=this.ctx,walk=n.moving?Math.sin(n.anim*12)*2:0,face=n.side||1;c.save();if(reflection){c.globalAlpha=.16;c.translate(0,GROUND*2+112);c.scale(1,-1);}this.pixel(goblin,x,y+walk,4,face);if(n.role==='hunter'){this.rect(x+face*21,y-35,4,28,'#926f3b');this.rect(x+face*25,y-30,3,16,'#c5b97b');}if(n.role==='worker'){this.rect(x+face*23,y-28,4,23,'#715d40');this.rect(x+face*23-9,y-34,20,10,'#849095');}if(n.role==='farmer'){this.rect(x-17,y-49,34,8,'#b19b68');this.rect(x-9,y-54,18,6,'#c3b17b');}if(n.role==='captain'){this.rect(x-15,y-48,30,12,'#879397');this.rect(x-22,y-31,16,23,'#7b898c');this.rect(x+20,y-40,4,30,'#c3c7b9');}c.restore();}
 player(reflection=false){const c=this.ctx,p=this.game.state.player,x=this.sx(p.x),walk=Math.abs(this.game.input.axis)>.14?Math.sin(this.t*17)*2:0,y=GROUND-p.fly+walk;const rows=p.mount==='wolf'?wolf:p.mount==='lizard'?lizard:p.mount==='bat'&&p.fly>8?bat:boar;c.save();if(reflection){c.globalAlpha=.16;c.translate(0,GROUND*2+112);c.scale(1,-1);}if(p.hit>0&&Math.floor(this.t*12)%2)c.globalAlpha*=.5;this.pixel(rows,x,y,5,p.face);this.pixel(goblin,x-p.face*4,y-32,4,p.face);this.rect(x-13,y-83,26,6,'#e7bd45');for(const a of [-12,-1,10])this.rect(x+a,y-89,4,7,'#e7bd45');c.fillStyle='#a34843';c.beginPath();c.moveTo(x-p.face*10,y-67);c.lineTo(x-p.face*56,y-52);c.lineTo(x-p.face*30,y-72);c.fill();if(!reflection&&p.stamina<99){this.rect(x-32,y+12,64,5,'#253234');this.rect(x-31,y+13,62*p.stamina/100,3,'#afc77a');}c.restore();}
 outside(){const c=this.ctx,img=this.image;const left=this.sx(-887);const right=this.sx(887);if(left>0||right<this.w){for(let x=-290-((this.camera%290)+290)%290;x<this.w+290;x+=290){c.drawImage(img,0,0,290,887,x,0,290,887);} // repeating quiet woodland; central plate masks this
 }c.drawImage(img,left,0,1774,887);}
 building(b){const x=this.sx(b.x);if(x< -180||x>this.w+180)return;const c=this.ctx,g=this.game;
 if(b.kind==='wall'){if(b.level&&b.hp>0){const h=35+b.level*18;for(let i=-24;i<25;i+=9){this.rect(x+i,GROUND-h,7,h,b.level===3?'#65757b':'#6e593e');this.rect(x+i+2,GROUND-h-7,3,10,b.level===3?'#97a0a0':'#aa8b55');}if(b.hp<b.max){this.rect(x-28,GROUND-h-18,56,5,'#26333b');this.rect(x-28,GROUND-h-18,56*b.hp/b.max,5,'#c28b58');}}else{this.rect(x-17,GROUND-10,5,10,'#786447');this.rect(x+12,GROUND-10,5,10,'#786447');}}
 if(b.kind==='tower'&&b.level){const h=80+b.level*20;this.rect(x-21,GROUND-h,5,h,'#66543e');this.rect(x+17,GROUND-h,5,h,'#66543e');this.rect(x-29,GROUND-h,58,8,'#94734c');this.rect(x-29,GROUND-h+20,58,5,'#94734c');this.npc({role:'hunter',side:Math.sign(b.x),anim:this.t,moving:false},x,GROUND-h);}
 if(b.kind==='farm'&&b.level){this.rect(x-70,GROUND-4,140,4,'#716245');for(let i=0;i<6;i++){const y=GROUND-7-(i%2)*6;this.rect(x-60+i*24,y-10,17,8,i%2?'#9b796e':'#c0ab80');this.rect(x-55+i*24,y-2,6,6,'#dfce9f');}this.rect(x+70,GROUND-46,4,46,'#8d784d');this.rect(x+62,GROUND-48,21,7,'#b6a577');}
 if(b.kind==='camp'){this.rect(x-40,GROUND-3,80,3,'#6b5638');c.fillStyle='#6d7450';c.beginPath();c.moveTo(x-48,GROUND-3);c.lineTo(x,GROUND-65);c.lineTo(x+43,GROUND-3);c.fill();this.rect(x-4,GROUND-43,10,40,'#2b3940');for(let i=0;i<b.stock;i++)this.npc({role:'idle',side:1,anim:0},x-80-i*30,GROUND);}
 if(b.kind==='portal'&&b.hp>0){this.rect(x-60,GROUND-92,18,92,'#4e6063');this.rect(x+42,GROUND-92,18,92,'#4e6063');this.rect(x-60,GROUND-103,120,20,'#66716b');this.rect(x-42,GROUND-85,84,85,'#1d2334');c.globalAlpha=.3+Math.sin(this.t*2)*.12;this.rect(x-32,GROUND-75,64,75,'#9b668b');c.globalAlpha=1;this.rect(x-45,GROUND-120,90*b.hp/b.max,5,'#b28b78');}
 if(b.kind==='rally'&&g.hall.level>=3){this.rect(x,GROUND-66,4,66,'#92744a');this.rect(x+(b.x<0?-24:4),GROUND-66,24,18,'#9c4f41');}
 if(b.kind.startsWith('tools')){const roles={toolsWorker:'worker',toolsHunter:'hunter',toolsFarmer:'farmer',toolsCaptain:'captain'},role=roles[b.kind];this.rect(x-24,GROUND-24,48,5,'#8e774e');this.rect(x-21,GROUND-20,4,20,'#6d583f');this.rect(x+18,GROUND-20,4,20,'#6d583f');if(role==='worker'){this.rect(x-2,GROUND-45,4,21,'#926d43');this.rect(x-11,GROUND-49,22,9,'#929a99');}if(role==='hunter'){this.rect(x+3,GROUND-49,4,25,'#9d7941');this.rect(x+7,GROUND-44,3,15,'#d2bc85');}if(role==='farmer'){this.rect(x-2,GROUND-48,4,24,'#9b794a');this.rect(x-12,GROUND-49,24,4,'#afaaa0');}if(role==='captain')this.rect(x-12,GROUND-45,25,20,'#8d999b');}
 if(b.kind==='tree'&&b.level){this.rect(x-8,GROUND-200,16,200,'#344b46');this.rect(x-55,GROUND-230,110,110,'#284941');this.rect(x-75,GROUND-185,150,60,'#335d49');}
 if(b.kind==='chest'&&b.level){this.rect(x-20,GROUND-25,40,25,'#926e35');this.rect(x-20,GROUND-20,40,5,'#d0a050');this.rect(x-4,GROUND-20,8,12,'#e2bc69');}
 if(b.kind==='hall'&&b.level>1){for(let i=0;i<b.level;i++)this.rect(x-80+i*90,GROUND-200,12,60,'#947b4b');this.text('◆'.repeat(b.level),x,GROUND-235,'#d6bf73',20);}
 if(b.job){this.rect(x-35,GROUND-60,70,6,'#283c40');this.rect(x-35,GROUND-60,70*(1-b.job.left/b.job.total),6,'#ceb46a');}
 }
 render(dt=1/60){this.t+=dt;this.fps=this.fps*.95+(1/Math.max(.001,dt))*.05;const c=this.ctx,s=this.game.state,p=s.player;const targetCamera=Math.max(-2900+this.w/2,Math.min(2900-this.w/2,p.x-200));this.camera+=(targetCamera-this.camera)*Math.min(1,dt*7);c.imageSmoothingEnabled=false;this.outside();
 for(const b of s.buildings)this.building(b);
 const merchants=[{role:'idle',x:-510,side:1},{role:'worker',x:355,side:-1},{role:'idle',x:630,side:-1}];for(const n of merchants)this.npc({...n,anim:this.t,moving:false});
 this.pixel(lizard,this.sx(610),GROUND,3.7,1);this.pixel(wolf,this.sx(710),GROUND,3.7,1);this.pixel(bat,this.sx(690),GROUND-80,2,1);
 for(const n of s.npcs){if(Math.abs(n.x-this.camera)<this.w/2+80){this.npc(n);this.npc(n,this.sx(n.x),GROUND,true);}}
 for(const e of s.enemies){if(Math.abs(e.x-this.camera)>this.w/2+100)continue;const palette={...C,g:e.type==='heavy'?'#787a6d':'#626285',G:e.type==='heavy'?'#a3a58e':'#9691af',b:'#3b3a51'};this.pixel(goblin,this.sx(e.x),GROUND, e.type==='heavy'?5:4,-e.side,palette);if(e.type==='ranged')this.rect(this.sx(e.x)+12,GROUND-39,5,30,'#9d7e63');}
 for(const a of s.shots)this.rect(this.sx(a.x),a.y,12,3,a.team==='friendly'?'#d3c49a':'#b991ad');for(const co of s.coins){const x=this.sx(co.x),y=GROUND-9-Math.sin(co.age*3)*3;this.rect(x-4,y-4,8,10,'#d7a430');this.rect(x-2,y-5,5,8,'#ffcf4f');}
 this.player(true);this.player();if(s.crown){this.rect(this.sx(s.crown.x)-10,GROUND-20,20,6,'#efc250');for(const x of [-9,-1,7])this.rect(this.sx(s.crown.x)+x,GROUND-25,3,6,'#efc250');}
 for(const e of s.effects)this.rect(this.sx(e.x)-8,e.y-8,16,16,'#e5c681');const tint=this.game.night?.42:this.game.phase==='黄昏'?.12:0;if(tint){c.fillStyle=`rgba(12,22,50,${tint})`;c.fillRect(0,0,this.w,this.h);}
 }
}
